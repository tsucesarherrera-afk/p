📋 Script SQL para Supabase (ejecutar en SQL Editor) 
CREATE TABLE IF NOT EXISTS personal_retirado (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    apellidos_nombres TEXT NOT NULL,
    cedula TEXT NOT NULL UNIQUE,
    genero TEXT, fecha_nacimiento DATE, edad INTEGER,
    telefono TEXT, correo TEXT, cargo_nominal TEXT,
    municipio TEXT, parroquia TEXT,
    ubicacion_administrativa TEXT, ubicacion_fisica TEXT,
    formacion_estudiando TEXT, observaciones TEXT,
    motivo_retiro TEXT NOT NULL,
    fecha_retiro DATE DEFAULT CURRENT_DATE,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- Permitir inserción/lectura pública (RLS)
ALTER TABLE personal_retirado ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Allow all" ON personal_retirado FOR ALL USING (true) WITH CHECK (true);
