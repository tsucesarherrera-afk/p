📋 Script SQL para crear tabla personal_retirado y restricciones 
-- Crear tabla para personal retirado
CREATE TABLE IF NOT EXISTS personal_retirado (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    apellidos_nombres TEXT NOT NULL,
    cedula TEXT NOT NULL UNIQUE,
    genero TEXT,
    fecha_nacimiento DATE,
    edad INTEGER,
    telefono TEXT,
    correo TEXT,
    cargo_nominal TEXT,
    municipio TEXT,
    parroquia TEXT,
    ubicacion_administrativa TEXT,
    ubicacion_fisica TEXT,
    formacion_estudiando TEXT,
    observaciones TEXT,
    motivo_retiro TEXT NOT NULL,
    fecha_retiro DATE DEFAULT CURRENT_DATE,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);
CREATE INDEX IF NOT EXISTS idx_personal_retirado_cedula ON personal_retirado(cedula);

-- UNIQUE en personal_enfermeria.cedula
DO $$
BEGIN
    IF NOT EXISTS (SELECT 1 FROM pg_constraint WHERE conname = 'personal_enfermeria_cedula_key') THEN
        ALTER TABLE personal_enfermeria ADD CONSTRAINT personal_enfermeria_cedula_key UNIQUE (cedula);
    END IF;
END $$;

-- UNIQUE en postulados.cedula
DO $$
BEGIN
    IF NOT EXISTS (SELECT 1 FROM pg_constraint WHERE conname = 'postulados_cedula_key') THEN
        ALTER TABLE postulados ADD CONSTRAINT postulados_cedula_key UNIQUE (cedula);
    END IF;
END $$;
